ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON); } try { // ── ping ──────────────────────────────────────────────────────────────── if (action === 'ping') { return respond({ ok: true, message: 'KCM Tracker Apps Script is running.' }); } // ── load ──────────────────────────────────────────────────────────────── if (action === 'load') { return respond(loadData()); } // ── saveField (small value — fits in one GET) ─────────────────────────── if (action === 'saveField') { var field = params.field; var data = params.data; var syncedAt = params.syncedAt || new Date().toISOString(); if (!field || data === undefined) return respond({ ok: false, error: 'Missing field or data' }); writeField(field, data, syncedAt); return respond({ ok: true }); } // ── chunkField (store one chunk row in the sheet) ─────────────────────── if (action === 'chunkField') { var session = params.session; var idx = params.idx; var data = params.data; if (!session || idx === undefined || data === undefined) return respond({ ok: false, error: 'Missing chunk params' }); writeChunk(session, parseInt(idx, 10), data); return respond({ ok: true }); } // ── assembleField (join sheet chunk rows → write field → delete chunks) ─ if (action === 'assembleField') { var session = params.session; var field = params.field; var total = parseInt(params.total, 10); var syncedAt = params.syncedAt || new Date().toISOString(); if (!session || !field || !total) return respond({ ok: false, error: 'Missing assemble params' }); var assembled = readAndDeleteChunks(session, total); if (assembled === null) return respond({ ok: false, error: 'Missing one or more chunks in sheet' }); writeField(field, assembled, syncedAt); return respond({ ok: true }); } // ── clearChunks (delete any leftover chunk rows) ─────────────────────── if (action === 'clearChunks') { var n = deleteAllChunkRows(); return respond({ ok: true, deleted: n }); } return respond({ ok: false, error: 'Unknown action: ' + action }); } catch (err) { return respond({ ok: false, error: err.toString() }); } } // ── Sheet helpers ──────────────────────────────────────────────────────────── function getSheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('KCM_Data'); if (!sheet) { sheet = ss.insertSheet('KCM_Data'); sheet.appendRow(['field', 'value', 'updatedAt']); } return sheet; } // Write or update a named field row function writeField(field, value, syncedAt) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); for (var r = 1; r <= lastRow; r++) { if (sheet.getRange(r, 1).getValue() === field) { sheet.getRange(r, 2).setValue(value); sheet.getRange(r, 3).setValue(syncedAt); return; } } sheet.appendRow([field, value, syncedAt]); } // Store a chunk as a row: field='__chunk____' function writeChunk(session, idx, data) { var sheet = getSheet(); var key = '__chunk__' + session + '__' + idx; var lastRow = sheet.getLastRow(); for (var r = 1; r <= lastRow; r++) { if (sheet.getRange(r, 1).getValue() === key) { sheet.getRange(r, 2).setValue(data); return; } } sheet.appendRow([key, data, '']); } // Read chunks 0..total-1, delete their rows, return assembled string (or null on missing) function readAndDeleteChunks(session, total) { var sheet = getSheet(); var lastRow = sheet.getLastRow(); // Build a map: idx → row number var rowMap = {}; var prefix = '__chunk__' + session + '__'; for (var r = 1; r <= lastRow; r++) { var key = sheet.getRange(r, 1).getValue(); if (key.indexOf(prefix) === 0) { var idx = parseInt(key.substring(prefix.length), 10); rowMap[idx] = r; } } var parts = []; for (var i = 0; i < total; i++) { if (rowMap[i] === undefined) return null; parts.push(sheet.getRange(rowMap[i], 2).getValue()); } // Delete chunk rows in reverse order to preserve row numbers var rows = Object.values(rowMap).sort(function(a,b){return b-a;}); for (var j = 0; j < rows.length; j++) { sheet.deleteRow(rows[j]); } return parts.join(''); } // Delete all leftover __chunk__ rows function deleteAllChunkRows() { var sheet = getSheet(); var lastRow = sheet.getLastRow(); var toDelete = []; for (var r = 1; r <= lastRow; r++) { if (String(sheet.getRange(r, 1).getValue()).indexOf('__chunk__') === 0) { toDelete.push(r); } } // Delete in reverse so row numbers stay valid for (var j = toDelete.length - 1; j >= 0; j--) { sheet.deleteRow(toDelete[j]); } return toDelete.length; } function loadData() { var sheet = getSheet(); var lastRow = sheet.getLastRow(); var result = { ok: true, data: {} }; for (var r = 1; r <= lastRow; r++) { var key = String(sheet.getRange(r, 1).getValue()); var val = sheet.getRange(r, 2).getValue(); // Skip chunk rows if (key.indexOf('__chunk__') === 0) continue; if (key && val) result.data[key] = val; } return result; } // doPost kept for any future use function doPost(e) { function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } return respond({ ok: false, error: 'POST not used by this version' }); }
